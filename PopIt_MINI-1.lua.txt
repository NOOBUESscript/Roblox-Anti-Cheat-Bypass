local function makeDraggable(frame, handle)
handle = handle or frame
local dragging = false
local dragInput
local dragStart
local startPos
local lastInput
local dragThreshold = 5
local hasMoved = false
local function updateDrag(input)
if not dragging then return end
local delta = input.Position - dragStart
local distance = math.sqrt(delta.X^2 + delta.Y^2)
if distance > dragThreshold then
hasMoved = true
end
if hasMoved then
frame.Position = UDim2.new( startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y
)
end
end
handle.InputBegan:Connect(function(input, gameProcessed)
if gameProcessed then return end
if input.UserInputType == Enum.UserInputType.MouseButton1 or
input.UserInputType == Enum.UserInputType.Touch then
dragging = true
hasMoved = false
dragStart = input.Position
startPos = frame.Position
lastInput = input
dragInput = input.InputChanged:Connect(function()
updateDrag(lastInput)
end)
end
end)
UserInputService.InputChanged:Connect(function(input, gameProcessed)
if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or
input.UserInputType == Enum.UserInputType.Touch) then
lastInput = input
updateDrag(input)
end
end)
handle.InputEnded:Connect(function(input, gameProcessed)
if input.UserInputType == Enum.UserInputType.MouseButton1 or
input.UserInputType == Enum.UserInputType.Touch then
dragging = false
if dragInput then
dragInput:Disconnect()
dragInput = nil
end
end
end)
return hasMoved
end
function getDisplayName(item)
if not item then return "Unknown" end
local attr = item:GetAttribute("DisplayName")
if attr and attr ~= "" then return tostring(attr) end
local sv = item:FindFirstChild("DisplayName")
if sv and sv:IsA("StringValue") and sv.Value ~= "" then return sv.Value end
local config = item:FindFirstChild("Configuration")
if config then
local dn = config:FindFirstChild("DisplayName")
if dn and dn:IsA("StringValue") and dn.Value ~= "" then return dn.Value end
end
if item:IsA("Tool") and item.ToolTip and item.ToolTip ~= "" then return item.ToolTip end
for _, gui in ipairs(item:GetDescendants()) do
if gui:IsA("BillboardGui") or gui:IsA("SurfaceGui") then
local lbl = gui:FindFirstChildOfClass("TextLabel")
if lbl and lbl.Text ~= "" then return lbl.Text end
end
end
return item.Name
end

local function findItem(search)
if not search then return nil end
search = string.lower(search:match("^%s*(.-)%s*$") or "")
local itemsFolder = ReplicatedStorage:FindFirstChild("Items")
if not itemsFolder then
showNotification("Error", "Items folder not found!", Color3.fromRGB(200, 50, 50))
return nil
end
for _, item in ipairs(itemsFolder:GetChildren()) do
local displayName = string.lower(getDisplayName(item))
if displayName == search then return item end
end
for _, item in ipairs(itemsFolder:GetChildren()) do
local realName = string.lower(item.Name)
if realName == search then return item end
end
for _, item in ipairs(itemsFolder:GetChildren()) do
local displayName = string.lower(getDisplayName(item))
if string.find(displayName, search, 1, true) then return item end
end
for _, item in ipairs(itemsFolder:GetChildren()) do
local realName = string.lower(item.Name)
if string.find(realName, search, 1, true) then return item end
end
return nil
end
function giveItemAsync(itemName)
local targetItem = findItem(itemName)
if not targetItem then
showNotification("Not Found", "Item not found: " .. itemName, Color3.fromRGB(200, 50, 50))
return
end
local displayName = getDisplayName(targetItem)
local realName = targetItem.Name
showNotification("Real Item Name Is:", realName, Color3.fromRGB(100, 150, 255))
showNotification("Giving 500", "Giving " .. displayName .. " x" .. AMOUNT, Color3.fromRGB(100, 200, 100))
local args = { targetItem, { questName = "Dumplings Quest", questRequiredItems = {}, prizeItem = "Rainbow Dumpling"
}
}
task.spawn(function()
for i = 1, AMOUNT do
pcall(function()
ReplicatedStorage
:WaitForChild("RemoteEvents")
:WaitForChild("GetItem")
:FireServer(unpack(args))
end)
end
task.wait(1)
showNotification("Complete", "Done! Gave " .. displayName .. " x" .. AMOUNT, Color3.fromRGB(50, 200, 100))
end)
end
function giveItem(itemName)
giveItemAsync(itemName)
end
local function handleCommand(message)
if message:sub(1, #PREFIX) ~= PREFIX then return end
local cmd = message:sub(#PREFIX + 1)
local lowerCmd = cmd:lower()
local itemName = cmd:match("^give me (.+)$")
if itemName then
giveItem(itemName)
return
end
itemName = cmd:match("^give (.+)$")
if itemName then
giveItem(itemName)
return
end
local checkItem = cmd:match("^check (.+)$")
if checkItem then
local found = findItem(checkItem)
if found then
local display = getDisplayName(found)
local real = found.Name
showNotification("Real Item Name Is:", real, Color3.fromRGB(100, 150, 255))
else
showNotification("Not Found", "Item not found: " .. checkItem, Color3.fromRGB(200, 50, 50))
end
return
end
if lowerCmd == "items all" or lowerCmd == "item all" then
createItemsGUI()
return
end
local searchTerm = cmd:match("^items (.+)$") or cmd:match("^item (.+)$")
if searchTerm and searchTerm ~= "all" then
createItemsGUI(searchTerm)
return
end
if lowerCmd == "items" or lowerCmd == "item" then
createItemsGUI()
return
end
if lowerCmd == "help" then
showNotification("Help", "Commands: :give {item}, :items, :check {item}", Color3.fromRGB(100, 150, 255))
end
end
